package com.meta.plugin;

import android.content.Context;
import android.view.*;
import android.widget.*;
import android.graphics.*;
import android.graphics.drawable.*;
import android.os.*;
import android.util.*;
import com.meta.app.bridge.ApiBridge;
import com.meta.app.core.PluginContract;

public class DynamicPlugin implements PluginContract {

    private TextView display;
    private StringBuilder expression = new StringBuilder();

    @Override
    public void onCreate(Context context, ViewGroup container, ApiBridge bridge) {
        LinearLayout root = new LinearLayout(context);
        root.setOrientation(LinearLayout.VERTICAL);
        root.setBackgroundColor(Color.parseColor("#121212"));
        root.setLayoutParams(new ViewGroup.LayoutParams(
            ViewGroup.LayoutParams.MATCH_PARENT,
            ViewGroup.LayoutParams.MATCH_PARENT));

        // display area
        display = new TextView(context);
        display.setText("0");
        display.setTextSize(48);
        display.setTextColor(Color.WHITE);
        display.setGravity(Gravity.END | Gravity.BOTTOM);
        display.setPadding(32, 32, 32, 32);
        display.setBackgroundColor(Color.parseColor("#1E1E1E"));
        display.setMaxLines(2);
        LinearLayout.LayoutParams displayParams = new LinearLayout.LayoutParams(
            ViewGroup.LayoutParams.MATCH_PARENT, 0, 0.3f);
        root.addView(display, displayParams);

        // button area
        GridLayout grid = new GridLayout(context);
        grid.setColumnCount(4);
        grid.setRowCount(5);
        LinearLayout.LayoutParams gridParams = new LinearLayout.LayoutParams(
            ViewGroup.LayoutParams.MATCH_PARENT, 0, 0.7f);
        grid.setLayoutParams(gridParams);

        String[] buttons = {
            "C", "(", ")", "÷",
            "7", "8", "9", "×",
            "4", "5", "6", "−",
            "1", "2", "3", "+",
            "0", ".", "⌫", "="
        };

        for (String btn : buttons) {
            Button b = new Button(context);
            b.setText(btn);
            b.setTextSize(22);
            b.setAllCaps(false);

            // styling
            if ("=".equals(btn)) {
                b.setBackgroundColor(Color.parseColor("#03DAC5"));
                b.setTextColor(Color.BLACK);
            } else if ("C".equals(btn) || "⌫".equals(btn)) {
                b.setBackgroundColor(Color.parseColor("#CF6679"));
                b.setTextColor(Color.WHITE);
            } else if ("÷×−+()".contains(btn)) {
                b.setBackgroundColor(Color.parseColor("#6200EE"));
                b.setTextColor(Color.WHITE);
            } else {
                b.setBackgroundColor(Color.parseColor("#2D2D2D"));
                b.setTextColor(Color.WHITE);
            }

            GridLayout.LayoutParams gp = new GridLayout.LayoutParams();
            gp.width = 0;
            gp.height = 0;
            gp.columnSpec = GridLayout.spec(GridLayout.UNDEFINED, 1, 1f);
            gp.rowSpec = GridLayout.spec(GridLayout.UNDEFINED, 1, 1f);
            gp.setMargins(4, 4, 4, 4);
            b.setLayoutParams(gp);

            b.setOnClickListener(v -> onButtonClick(btn));
            grid.addView(b);
        }

        root.addView(grid);
        container.addView(root);
    }

    private void onButtonClick(String btn) {
        switch (btn) {
            case "C":
                expression.setLength(0);
                display.setText("0");
                break;
            case "⌫":
                if (expression.length() > 0) {
                    expression.deleteCharAt(expression.length() - 1);
                    display.setText(expression.length() > 0 ? expression.toString() : "0");
                }
                break;
            case "=":
                try {
                    String expr = expression.toString()
                        .replace("×", "*").replace("÷", "/").replace("−", "-");
                    double result = eval(expr);
                    if (result == Math.floor(result) && !Double.isInfinite(result)) {
                        display.setText(String.valueOf((long) result));
                    } else {
                        display.setText(String.valueOf(result));
                    }
                    expression.setLength(0);
                    expression.append(display.getText());
                } catch (Exception e) {
                    display.setText("Error");
                    expression.setLength(0);
                }
                break;
            default:
                expression.append(btn);
                display.setText(expression.toString());
                break;
        }
    }

    private double eval(String expr) {
        return new Object() {
            int pos = -1, ch;
            void nextChar() { ch = (++pos < expr.length()) ? expr.charAt(pos) : -1; }
            boolean eat(int charToEat) {
                while (ch == ' ') nextChar();
                if (ch == charToEat) { nextChar(); return true; }
                return false;
            }
            double parse() { nextChar(); double x = parseExpr(); return x; }
            double parseExpr() {
                double x = parseTerm();
                for (;;) {
                    if (eat('+')) x += parseTerm();
                    else if (eat('-')) x -= parseTerm();
                    else return x;
                }
            }
            double parseTerm() {
                double x = parseFactor();
                for (;;) {
                    if (eat('*')) x *= parseFactor();
                    else if (eat('/')) x /= parseFactor();
                    else return x;
                }
            }
            double parseFactor() {
                if (eat('+')) return +parseFactor();
                if (eat('-')) return -parseFactor();
                double x;
                int start = this.pos;
                if (eat('(')) { x = parseExpr(); eat(')'); }
                else if ((ch >= '0' && ch <= '9') || ch == '.') {
                    while ((ch >= '0' && ch <= '9') || ch == '.') nextChar();
                    x = Double.parseDouble(expr.substring(start, this.pos));
                } else { throw new RuntimeException("Unexpected: " + (char)ch); }
                return x;
            }
        }.parse();
    }

    @Override
    public void onDestroy() {}

    @Override
    public String getTitle() {
        return "Calculator";
    }
}
